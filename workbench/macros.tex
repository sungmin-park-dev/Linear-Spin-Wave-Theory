% Project macros for the research notes in workbench/notes/ (research-workspace inputs this file
% after the LaTeX template "research-note"). They carry the definitions of docs/nbcp/preamble.tex
% that the notes duplicated from the NBCP manuscript on 2026-10-05 still use.

% Reference to a place in another research note (the notes are compiled separately)
\providecommand{\nbcpnoteref}[2]{#2 in research note \emph{#1}}

% Image scaling used by the figure environments
\usepackage{graphicx}
\makeatletter
\newsavebox\nbcp@box
\providecommand*\nbcpbounded[1]{%
  \sbox\nbcp@box{#1}%
  \Gscale@div\@tempa{\textheight}{\dimexpr\ht\nbcp@box+\dp\nbcp@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\nbcp@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\nbcp@box}%
  \else\usebox{\nbcp@box}%
  \fi
}
\makeatother
