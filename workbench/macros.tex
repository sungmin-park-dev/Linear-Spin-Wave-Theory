\usepackage{graphicx}
\makeatletter
\newsavebox\nbcp@box
\providecommand*\nbcpbounded[1]{%
  \sbox\nbcp@box{#1}%
  \Gscale@div\@tempa{\textheight}{\dimexpr\ht\nbcp@box+\dp\nbcp@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\nbcp@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\nbcp@box}%
  \else\usebox{\nbcp@box}%
  \fi
}
\makeatother
